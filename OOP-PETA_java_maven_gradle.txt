import presentationlayer.BaseMenu;

import presentationlayer.RequestorMenu;
import presentationlayer.SupportAgentMenu;

public class CampusHelpDeskTicketingSystem {

    private static Repository repository;
    private static Scanner input;
    
    public static void main(String[] args) {
        repository = new Repository();
        input = new Scanner(System.in);
        seedData(repository);
        run();
        input.close();
    } // Initialize Repository and Scanner

    public static void run() {
        while(true) {
            LoginMenu loginMenu = new LoginMenu(input, repository);
            loginMenu.run();
            User user = loginMenu.getAuthenticatedUser();
            if (user == null ) { break; }
            launchMenu(user).run();
            System.out.println("You have been logged Out");
        }
    } // Main Loop of whole System

    public static BaseMenu launchMenu(User user) {
        if (user == null) throw new IllegalArgumentException("User cannot be null");
        if (user instanceof Admin admin)         return new AdminMenu(admin, input, repository);
        if (user instanceof SupportAgent agent)  return new SupportAgentMenu(agent, input, repository);
        if (user instanceof Requestor requestor) return new RequestorMenu(requestor, input, repository);
        throw new IllegalStateException("No menu for " + user.getRole());
    }

    public static void seedData(Repository repo) {

        if (repo == null) { return; }

        // 1. Seed Users (Admin, SupportAgents, Student, Faculty)
        Admin admin = new Admin("ADM-001", "Alice", "Admin", "alice@univ.edu", "pass123", repo, "EMP-100", "IT", "ext111");

package entitylayer;

import datalayer.Repository;

import java.util.LinkedHashMap;

public class Admin extends User {

    private String employeeId;
    private String managedDepartment;
    private String deskExtension;

    public Admin(String userId, String firstName, String lastName, String email, String password, Repository repository, String employeeId, String managedDepartment, String deskExtension) {
        super(userId, firstName, lastName, email, password, repository);
        this.employeeId = employeeId;
        this.managedDepartment = managedDepartment;
        this.deskExtension = deskExtension;
    }

    @Override
    public String getRole() {
       return "ADMIN";
    }

    @Override
    public Ticket getTicket(String ticketId) {
        Ticket ticket =getRepository().findTicketById(ticketId);
        if (ticket == null) {
            throw new IllegalArgumentException("Ticket " + ticketId + " was not found.");
        }
        return ticket;
    }

    // Getters
    public String getEmployeeId() { return employeeId; }
    public String getManagedDepartment() { return managedDepartment; }
    public String getDeskExtension() { return deskExtension;  }
    
    // Setters
    public void setEmployeeId(String employeeId) { this.employeeId = employeeId; }
    public void setManagedDepartment(String managedDepartment) { this.managedDepartment = managedDepartment; }
    public void setDeskExtension(String deskExtension) { this.deskExtension = deskExtension; }

    public void reassignTicket(String ticketId, String agentId, String reason) {
        if (agentId == null || reason == null) {
            throw new IllegalArgumentException("AgentId and reason are required");
        }
        Ticket ticket = getTicket(ticketId);
        User user = getRepository().findUserById(agentId);

        if (!(user instanceof SupportAgent agent)) { 
            throw new IllegalArgumentException("User with " + agentId + " is not an agent.");
        }
        ticket.reassignTicket(agent, this, reason);
    } 

    public void rejectTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId);
        ticket.changeState("CANCELLED", this, reason);
    }

    public ArrayList<Ticket> getAllTickets() {
        ArrayList<Ticket> allTickets = getRepository().getAllTickets();
        if (allTickets == null) { return new ArrayList<>(); }
        return allTickets;
    }

    public ArrayList<User> getAllUsers() {
        ArrayList<User> allUsers = getRepository().getAllUsers();
        if (allUsers == null) { return new ArrayList<>(); }
        return allUsers;
    }

    public ArrayList<Ticket> filterTickets(String field, String value) {
        ArrayList<Ticket> result = new ArrayList<>();
        if (field == null || value == null) { return result; }

        ArrayList<Ticket> allTickets = getAllTickets();
        for (Ticket ticket : allTickets) {
            String actual = switch (field.toUpperCase()) {
                case "ID"        -> ticket.getTicketId();
                case "STATE"     -> ticket.getState();
                case "CATEGORY"  -> ticket.getCategory();
                case "PRIORITY"  -> ticket.getPriority();
                case "USER"      -> ticket.getSubmittedBy().getUserId();
                case "AGENT"     -> ticket.getAssignedAgent() == null ? "UNASSIGNED" : ticket.getAssignedAgent().getUserId() ; 
                default          -> null;
            };
            if (actual != null && actual.equalsIgnoreCase(value.trim())) {
                result.add(ticket);
            }
        }
        return result;
    }

    public ArrayList<User> filterUsers(String field, String value) {
        ArrayList<User> result = new ArrayList<>();
        if (field == null || value == null) { return result; }

        ArrayList<User> allUsers = getAllUsers();
        for (User user : allUsers ) {
            if (user == null) { continue; }
            String actual = switch(field.toUpperCase()) {
                case "ID"             -> user.getUserId();
                case "ROLE"           -> user.getRole();
                case "NAME"           -> user.getFullName();
                case "SPECIALIZATION" -> user instanceof SupportAgent supportAgent ? supportAgent.getSpecialization() : null;
                default               -> null;
            };
            if (actual  != null && actual.equalsIgnoreCase(value.trim())) {
                result.add(user);
            }
        }
        return result;
    }

    public LinkedHashMap<String, Integer> getSummaryCounts() {
        LinkedHashMap<String, Integer> counts = new LinkedHashMap<>();
        counts.put("totalTickets", getAllTickets().size()); // TICKET
        counts.put("totalUsers", getAllUsers().size()); //  / USERS
       
        String[] states = {"OPEN", "IN_PROGRESS", "RESOLVED", "CLOSED", "CANCELLED"};
        for (String state : states) { counts.put(state, filterTickets("STATE", state).size()); } // States
        String[] priorities = {"LOW", "MEDIUM", "HIGH", "URGENT"};
        for (String priority : priorities ) { counts.put(priority, filterTickets("PRIORITY", priority).size()); } // PRIORITY
        String[] categories = {"FACILITIES_MAINTENANCE", "SOFTWARE_SYSTEMS", "NETWORK", "HARDWARE", "ACCOUNT_SECURITY"};
        for (String category : categories) { counts.put(category, filterTickets("CATEGORY", category).size()); } // CATEGORY

        counts.put("UnassignedTickets", filterTickets("AGENT", "UNASSIGNED").size()); // Others
        return counts;
    }

}

public class Comment {

    private final User user;

    public Comment(User user, String message) {
        this.commentId = generateCommentId();
        this.user = user;
        this.message = message;
        this.timestamp = LocalDateTime.now();
    }

    public String getCommentId() {return commentId; } 
    public User getAuthor() {return user;}
    public String getAuthorName() { return user.getFullName(); }
    public String getMessage() {return message;}
    public LocalDateTime getTimestamp() { return timestamp;}
    public String getTime() { return getTimestamp().format(java.time.format.DateTimeFormatter.ofPattern("MM-dd HH:mm")); }

    private String generateCommentId() {
        commentCounter++;
        return String.format("CM-%03d", commentCounter);
    } // called whenever a new comment is created

}

package entitylayer;

import datalayer.Repository;
import java.util.ArrayList;

public class Requestor extends User {

    public Requestor(String userId, String firstName, String lastName, String email, String password, Repository repository) {
        super(userId, firstName, lastName, email, password, repository);
    }

    @Override
    public String getRole() {
        return "REQUESTOR";
    }

    @Override
    public Ticket getTicket(String ticketId) {
        Ticket ticket = getRepository().findTicketById(ticketId);
        if (ticket == null) { 
            throw new IllegalArgumentException("Ticket " + ticketId +" was not found."); 
        }
        if (ticket.getSubmittedBy() == null || !ticket.getSubmittedBy().getUserId().equalsIgnoreCase(getUserId())) { 
            throw new SecurityException("You can only view your own tickets."); 
        }
        return ticket;  
    } // throws an exception if invalid Ticket ID / if you do not own ticket

    public Ticket createTicket(String priority, String category, String subject, String description) {
        return new Ticket(this, null, priority, category, subject, description);
    } 

    public Comment createComment(String message){
        return new Comment(this, message);
    }
    
    public void submitTicket(Ticket ticket) {
        if (ticket == null) { 
            throw new IllegalArgumentException("Cannot submit an empty Ticket"); 
        }
        assignTicket(ticket); // tickets stays unassigned if no agent matches
        getRepository().addTicket(ticket);
     }
     
    public void cancelTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.changeState("CANCELLED", this, reason);
     }
     
    public void reopenTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.changeState("IN_PROGRESS", this, reason);
     }
     
    public void addComment(String ticketId, String message) {
        Ticket ticket = getTicket(ticketId);
        ticket.addComment(createComment(message));
    }

    public void closeTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.changeState("CLOSED", this, reason);
    }

    public ArrayList<Ticket> getMyTickets() {
        ArrayList<Ticket> myTickets = new ArrayList<>();
        ArrayList<Ticket> allTickets = getRepository().getAllTickets();
        if (allTickets == null ) { return myTickets; }
        for (Ticket ticket : allTickets) {
                if (ticket.getSubmittedBy().getUserId().equalsIgnoreCase(getUserId())) {
                    myTickets.add(ticket);
                }
            }
        return myTickets;
    }
    
    public ArrayList<Ticket> getActiveTickets() {
        ArrayList<Ticket> myTickets = new ArrayList<>();
        for (Ticket ticket : getMyTickets()) {
            String state = ticket.getState();
            if (state.equalsIgnoreCase("OPEN") || state.equalsIgnoreCase("IN_PROGRESS")) {
                myTickets.add(ticket);
            }
        }
        return myTickets;
     } 

    private boolean assignTicket(Ticket ticket) {
        ArrayList<User> users = getRepository().getAllUsers();
        if (users == null) { return false; }

        SupportAgent bestAgent = null;
        int fewestActiveTickets = Integer.MAX_VALUE;

        for (User user : users) {
            if (!(user instanceof SupportAgent agent)) { continue; }
            if (agent.getSpecialization() == null ) { continue; }
            if (!agent.getSpecialization().equalsIgnoreCase(ticket.getCategory())) { continue; }
            
            int activeCount = agent.getActiveTickets().size();
            if (activeCount < fewestActiveTickets) {
                fewestActiveTickets = activeCount;
                bestAgent = agent;
            }
        }

        if (bestAgent != null) { 
            ticket.setAssignedAgent(bestAgent);
            return true;
        } 
        return false;
    } 

}

            }

import java.util.ArrayList;

public class Ticket {

    private final Requestor submittedBy;

    private String priority;
    private String category;
    private String subject;
    private String description;
    private final ArrayList<Comment> comments;
    private final ArrayList<TicketLogEntry> ticketLogEntries;
    private static int ticketCounter = 0;

    public Ticket(Requestor submittedBy, SupportAgent assignedTo, String priority, String category, String subject, String description) {
        this.ticketId = generateTicketId();
        this.submittedBy = submittedBy;
        this.assignedAgent = assignedTo;
        this.priority = priority;
        this.category = category;
        this.subject = subject;
        this.dateSubmitted = LocalDateTime.now();
        this.state = "OPEN";
        this.description = description;
        comments = new ArrayList<>();
        ticketLogEntries = new ArrayList<>();
        ticketLogEntries.add(new TicketLogEntry("OPEN", "OPEN", submittedBy.getUserId(), submittedBy.getFirstName(), "Ticket Created"));
    }
    
    // Getters
    public String getTicketId() { return ticketId; }
    public Requestor getSubmittedBy() { return submittedBy;}

    public String getPriority() {return priority;}

    public String getState() {return state;}

    public String getDescription() {return description;}
    public ArrayList<Comment> getComments() {return new ArrayList<>(comments);}
    public ArrayList<TicketLogEntry> getTicketLogEntries() {return new ArrayList<>(ticketLogEntries);}

    // setters
    public void setAssignedAgent(SupportAgent agent) {this.assignedAgent = agent;}
    public void setCategory(String category) {this.category = category; }
    public void setSubject(String subject) { this.subject = subject; }
    public void setDescription(String description) {this.description = description;}

    public void addComment(Comment comment) {
        if (comment == null) { 
            throw new IllegalArgumentException("Comment cannot be empty"); 
        }
        if (state.equalsIgnoreCase("CLOSED") || state.equalsIgnoreCase("CANCELLED")) { 
            throw new IllegalStateException("Ticket " + ticketId + " is " + getState() + ". Comments are not allowed on Closed or Cancelled tickets."); 
        }
        comments.add(comment);
    }

    public void changeState(String newState, User changedBy, String reason) {
        if (newState == null  || changedBy == null || reason == null) { 
            throw new IllegalArgumentException("State, user, and reason are required."); 
        } 
        if (getState().equalsIgnoreCase("CANCELLED")) { 
            throw new IllegalStateException("Ticket " + ticketId + " is Cancelled and cannot be changed."); 
        }
        if (!isValidTransition(state, newState)) {
            throw new IllegalStateException("Cannot move ticket " + ticketId + " from " + getState() + " to " + newState.toUpperCase() + ".");
        }
        String oldState = this.state;
        this.state = newState.toUpperCase();
        ticketLogEntries.add(new TicketLogEntry(oldState, this.state, changedBy.getUserId(), changedBy.getFirstName(), reason.trim()));
    }
    
    public boolean reassignTicket(SupportAgent newAgent, User changedBy, String reason) {
        if (newAgent == null || changedBy == null || reason == null) { 
            throw new IllegalStateException("Agent, user, and reason cannot be empty"); 
        }
        if (!(state.equalsIgnoreCase("OPEN") || state.equalsIgnoreCase("IN_PROGRESS"))) { 
            throw new IllegalStateException("Ticket " + ticketId + " only OPEN or IN-PROGRESS tickets can be reassigned.");
        }
        if (newAgent == this.assignedAgent) { 
            throw new IllegalArgumentException("Ticket " + ticketId + " is already assigned to " + newAgent.getFirstName() + "."); 
        }

        String oldAgentName = getAssignedAgent().getFullName();
        String newAgentName = newAgent.getFullName();
        setAssignedAgent(newAgent);
        ticketLogEntries.add(new TicketLogEntry(oldAgentName, newAgentName, changedBy.getUserId(), changedBy.getFirstName(), reason));
        return true;
    }

    private boolean isValidTransition(String oldState, String newState) {
        return (oldState.equalsIgnoreCase("OPEN") && newState.equalsIgnoreCase("IN_PROGRESS")) ||
               (oldState.equalsIgnoreCase("IN_PROGRESS") && newState.equalsIgnoreCase("RESOLVED")) ||
               (oldState.equalsIgnoreCase("RESOLVED") && newState.equalsIgnoreCase("CLOSED")) ||
               (oldState.equalsIgnoreCase("OPEN") && newState.equalsIgnoreCase("CANCELLED")) ||
               (oldState.equalsIgnoreCase("IN_PROGRESS") && newState.equalsIgnoreCase("CANCELLED")) ||
               (oldState.equalsIgnoreCase("RESOLVED") && newState.equalsIgnoreCase("IN_PROGRESS"));
    }    
    
    public void changePriority(String newPriority, String oldPrority,  String reason, User changedBy) {
        if (newPriority == null || changedBy == null) { 
            throw new IllegalArgumentException("Priority and User are required"); 
        }
        if (state.equalsIgnoreCase("CLOSED") || state.equalsIgnoreCase("CANCELLED")) { 
            throw new IllegalArgumentException("Cannot changed the priority on a ticket that is " + getState()); 
        }
        newPriority = newPriority.toUpperCase();
        if (!(newPriority.equals("LOW") || newPriority.equals("MEDIUM") || newPriority.equals("HIGH") || newPriority.equals("URGENT"))) { 
            throw new IllegalArgumentException("Invalid priority: " + newPriority + ". Choose URGENT, HIGH, MEDIUM, or LOW.");
        }
        if (newPriority.equalsIgnoreCase(getPriority())) { 
            throw new IllegalArgumentException("Ticket " + ticketId + " is already " + priority + " priority"); 
        } 
        ticketLogEntries.add(new TicketLogEntry(oldPrority, newPriority, changedBy.getUserId(), changedBy.getFirstName(), reason));
        this.priority = newPriority;
    } 

    private String generateTicketId() {
        ticketCounter++;
        return String.format("TK-%03d", ticketCounter);
    } // called whenever a new tikcet is created

Roles

Requestor (Student or Faculty): submits tickets, comments, cancels, reopens and closes their own tickets.
Support Agent: works tickets assigned to them, updates status, changes priority and comments.
Admin: views everything, filters, reassigns and rejects tickets, and sees the summary report.

Ticket lifecycle: OPEN → IN_PROGRESS → RESOLVED → CLOSED, with CANCELLED as a side exit.

The requestor can cancel (OPEN or IN_PROGRESS), reopen (RESOLVED only) and close (RESOLVED).
The agent can start work, resolve, or undo a mistaken resolve.
The admin can reject (OPEN or IN_PROGRESS) and reassign.
Every state change is logged with who, when and why. Reasons are required.

Rules we locked today

New tickets go to the matching-specialization agent with the fewest active tickets. If none match, the ticket stays unassigned.
There’s no agent ticket cap, no SLA and no auto-close.
Requestors choose priority when submitting. Agents can raise or lower it later.
Comments are blocked on CLOSED and CANCELLED tickets, and admins can’t comment.
IDs are TK-001 (tickets), CM-001 (comments), LG-001 (log entries), and REQ-, AGT- and ADM- for users.

Code cleanup: we reviewed all the entity classes and fixed bugs in reassign, reject and priority changes. IDs are now generated inside each entity, and the repository only stores data.

Menu layer (built so far)

BaseMenu: the parent of every menu. It runs the console loop (run() / stop()) and handles safe input (readInt, readLine keep asking until the input is valid). It also has truncate(...), which clips long text like FACILITIES_MAINTENANCE so table columns stay aligned.
Seeder (seedData): fills the app on startup with sample users (admin, agents, a student, a faculty member) plus tickets, comments and history, including edge cases.
RequestorMenu: the “my tickets” table is done, with aligned columns and names resolved from IDs.

How LoginMenu works

The program starts and seedData loads the sample data.
LoginMenu asks for a user ID and password.
It checks them against the users in the repository, and wrong credentials are rejected.
If valid, it looks at the user’s role and opens the matching menu: Requestor, Support Agent or Admin.
That role menu runs until the user picks Logout.
Logout returns to LoginMenu, which shows the login prompt again so someone else can sign in.
