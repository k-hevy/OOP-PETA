public class CampusHelpDeskTicketingSystem {

    // Serves as the main application class, it is the start of the program. It initializes the repository and scanner, 
    // creates sample users and tickets, handles log in loop, and opens correct menu based on authenticated user role

    private static Repository repository;
    private static Scanner input;
    
    // Initializes Repository and Scanner, the main entry point of the system.
    public static void main(String[] args) {
        repository = new Repository();
        input = new Scanner(System.in);
        seedData(repository);
        run();
        input.close();
    } // Initialize Repository and Scanner 

    // Executes the main application runtime loop, managing user login, session handling, 
    // menu launching, and logout sequences.
    public static void run() {
        while(true) {
            LoginMenu loginMenu = new LoginMenu(input, repository);
            loginMenu.run();
            User user = loginMenu.getAuthenticatedUser();
            if (user == null ) { break; }
            try {
                getMenu(user).run();
            } catch (IllegalStateException e) {
                System.out.println("Error: " + e.getMessage());
            }
            System.out.println("You have been logged Out");
        }
    } // Main Loop of whole System


    // Instantiates and returns the appropriate menu view based on the authenticated user's role.
    // @param user The currently authenticated user.
    // @return The corresponding BaseMenu implementation for the user's role.
    // @throws IllegalArgumentException if the provided user is null.
    // @throws IllegalStateException if no menu mapping exists for the user's role.
    public static BaseMenu getMenu(User user) {
        
        if (user == null) throw new IllegalArgumentException("User cannot be null");
        if (user instanceof Admin admin)         return new AdminMenu(admin, input, repository);
        if (user instanceof SupportAgent agent)  return new SupportAgentMenu(agent, input, repository);
        if (user instanceof Requestor requestor) return new RequestorMenu(requestor, input, repository);
        throw new IllegalStateException("No menu for " + user.getRole());

    }

package datalayer;

import entitylayer.Ticket;
import entitylayer.User;
import java.util.ArrayList;

public class Repository {

    // It acts as the central storage area that other class can use to add, search, and retrieve users and tickets

    private final ArrayList<User> users;
    private final ArrayList<Ticket> tickets;
    
    //Constructs a new Repository instance and initializes internal user and ticket lists.
    public Repository() {
        users = new ArrayList<>();
        tickets = new ArrayList<>();
    }

    // Receives a new user and adds a new user to the repository
    public void addUser(User user) {
        users.add(user);
    }

    // Receives an id and searches for the user with that id and returns a user or null if none.
    public User findUserById(String targetId) {
        for (User user : users) {
            if (user.getUserId().equalsIgnoreCase(targetId)) {
                return user;
            }
        }
        return null; // no user with this id exists
    }

    // Returns a copy of all users in the system
    public ArrayList<User> getAllUsers() {
        return new ArrayList<>(users);
    } 

    // Receives a ticket and adds a new ticket to the repository
    public void addTicket(Ticket ticket) {
        tickets.add(ticket);
    }

    // Receives an id and searches for the ticket with that id and returns a ticket or null if none.
    public Ticket findTicketById(String targetId) {
         for (Ticket ticket : tickets) {
            if (ticket.getTicketId().equalsIgnoreCase(targetId)) {
                return ticket;
            }
        }
        return null; // no ticket with this id exists
    }

    // Returns a copy of all tickets in the system
    public ArrayList<Ticket> getAllTickets() {
        return new ArrayList<>(tickets);
    }

}

package entitylayer;

import java.time.LocalDateTime;

public class Comment {

    // Represents a comment added to a ticket, it stores the comment’s ID, author, message, date and time it was created. 
    // This allows the Requestor and Support Agent to communicate properly.

    private final String commentId;
    private final User author;
    private final String message;
    private final LocalDateTime timestamp;
    private static int commentCounter = 0;

    // Constructs a new Comment instance with author and message
    public Comment(User author, String message) {
        this.commentId = generateCommentId();
        this.author = author;
        this.message = message;
        this.timestamp = LocalDateTime.now();
    }

    public String getCommentId() {return commentId; } 
    public User getAuthor() {return author;}
    public String getAuthorName() { return author.getFullName(); }
    public String getMessage() {return message;}
    public LocalDateTime getTimestamp() { return timestamp;}

    // called whenever a new comment is created
    private String generateCommentId() {
        commentCounter++;
        return String.format("CM-%03d", commentCounter);
    } 
}

package entitylayer;

import datalayer.Repository;
import java.util.ArrayList;

public class Requestor extends User {

    // It represents users who submit help desk tickets, such as students and faculty members. Requestors can only access their own tickets. 
    // Has the capability to create, submit, delete, reopen, close and add Comments on tickets. 

    // Constructs a Requestor instance
    public Requestor(String userId, String firstName, String lastName, String email, String password, Repository repository) {
        super("REQUESTOR",userId, firstName, lastName, email, password, repository);
    }

    // Accepts a ticket id and finds it. throws an exception if invalid Ticket ID / if you do not own ticket
    @Override
    public Ticket getTicket(String ticketId) {
        Ticket ticket = getRepository().findTicketById(ticketId);
        if (ticket == null || ticket.getState().equalsIgnoreCase("DELETED")) { 
            throw new IllegalArgumentException("Ticket " + ticketId +" was not found."); 
        } // Throw an error if you try to acces a ticket that doesnt exist
        if (!ticket.getSubmittedBy().getUserId().equalsIgnoreCase(getUserId())) { 
            throw new SecurityException("Ticket " + ticketId + " is not yours."); 
        } // Throw an Error if you try to access a ticket that isnt your
        return ticket;  
    } 

    // Returns all tickets submitted by the requestor except Deleted ones
    @Override
    public ArrayList<Ticket> getTickets() {
       ArrayList<Ticket> myTickets = new ArrayList<>();
        ArrayList<Ticket> allTickets = getRepository().getAllTickets();
        for (Ticket ticket : allTickets) {
                if (ticket.getSubmittedBy().getUserId().equalsIgnoreCase(getUserId()) && !ticket.isDeleted()) {
                    myTickets.add(ticket);
                }
            }
        return myTickets;
    } 

    // Receives a priority, category, subject and description and Create a tickets then returns it.
    public Ticket createTicket(String priority, String category, String subject, String description) {
        return new Ticket(this, null, priority, category, subject, description);
    } 
    
    // after creating a ticket it calls ticket methods to auto-assign to agent and store in the repository
    public void submitTicket(Ticket ticket) {
        ticket.autoAssignAgent(getRepository().getAllUsers());
        getRepository().addTicket(ticket);
     } 
     
    // Receives a ticketID and reason, find the ticket. Only allows deletion if a ticket if it is OPEN - IN-PROGRESS then calls ticket methods
    public void deleteTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.requireActive(); // The ticket should be OPEN / IN_PROGRESS
        ticket.deleteTicket(this, reason); 
     } 
     
    // Receives a ticketID and reason, find the ticket. Reopens a ticket only if its RESOLVED
    public void reopenTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.reopenTicket(this, reason);
    } 
     
    // Receives a ticketID and reason, find the ticket. Adds a comment to the specified ticket
    public void addComment(String ticketId, String message) {
        Ticket ticket = getTicket(ticketId);
        ticket.addComment(new Comment(this, message));
    }

    // Receives a ticketID and reason, find the ticket. Close a ticket that is resolved and
    public void closeTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.closeTicket(this, reason);
    } // Close aticket that is resolved

    

}

        } catch (IllegalArgumentException | SecurityException | IllegalStateException e) {
            printError(e.getMessage());
        }
        waitForEnter();
    }

    // Handles updating a ticket's priority 
    public void updateTicketPriority() {
        printHeader("Update Ticket Priority");
        String ticketId = readLine("Enter ticket ID: ", false);
        try {
            Ticket ticket = supportAgent.getTicket(ticketId);
            ticket.requireActive();
            String oldPriority = ticket.getPriority();
            String newPriority = readPriority();
            if (confirm("Change ticket priority from " + oldPriority + " to " + newPriority + "?")) {
                String reason = readLine("Enter a reason: ", false);
                supportAgent.updatePriority(ticketId, newPriority, reason);
                System.out.println("Success! Ticket priority has been changed from " + oldPriority  + " to " + newPriority);
            } else { 
                System.out.println("Ticket priority change Cancelled.");
            }
        } catch (IllegalArgumentException | SecurityException | IllegalStateException e) {
            printError(e.getMessage());
        }
        waitForEnter();
    }

    // Handles adding a comment to an assigned ticket
    public void addComment() {
        printHeader("Add Comment");
        String ticketId = readLine("Enter ticket ID: ", false);
        try {
            Ticket ticket = supportAgent.getTicket(ticketId);
            ticket.requireNotFinished();
            String message = readLine("Enter comment: ", false);
            supportAgent.addComment(ticketId, message);
            System.out.println("Success! Comment added to " + ticket.getTicketId() + ".");
        } catch (IllegalArgumentException | SecurityException | IllegalStateException e) {
           printError(e.getMessage());
        }
        waitForEnter();
    }

    // Prompts the agent to select an allowed ticket state transition
    private String readAgentState() { 
        while(true) {
            System.out.println("");
            System.out.println("[1] OPEN");
            System.out.println("[2] IN-PROGRESS");
            System.out.println("[3] RESOLVED");
            System.out.println("");
            int choice = readInt("choice: ", 1, 3);
            switch (choice) {
                case 1 -> { return "OPEN"; }
                case 2 -> { return "IN_PROGRESS"; }
                case 3 -> { return "RESOLVED"; }
            }
        }
    }
