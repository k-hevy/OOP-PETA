                System.out.println("Error: " + e.getMessage());

package datalayer;

import entitylayer.Ticket;
import entitylayer.User;
import java.util.ArrayList;

public class Repository {

    private final ArrayList<User> users;
    private final ArrayList<Ticket> tickets;
    

    public Repository() {
        users = new ArrayList<>();
        tickets = new ArrayList<>();
    }

    public void addUser(User user) {
        users.add(user);
    }

    public User findUserById(String targetId) {
        for (User user : users) {
            if (user.getUserId().equalsIgnoreCase(targetId)) {
                return user;
            }
        }
        return null; // no user with this id exists
    }

    public ArrayList<User> getAllUsers() {
        return new ArrayList<>(users);
    } 

    public void addTicket(Ticket ticket) {
        tickets.add(ticket);
    }

    public Ticket findTicketById(String targetId) {
         for (Ticket ticket : tickets) {
            if (ticket.getTicketId().equalsIgnoreCase(targetId)) {
                return ticket;
            }
        }
        return null; // no ticket with this id exists
    }

    public ArrayList<Ticket> getAllTickets() {
        return new ArrayList<>(tickets);
    }

}

        ArrayList<Ticket> result = new ArrayList<>();
        
        ArrayList<Ticket> allTickets = getTickets(); // gets all tickets
        for (Ticket ticket : allTickets) { // EX FIELD = CATEGORY, VALUE = MAINTENANCE
            String actual = switch (field.toUpperCase()) {
                case "ID"        -> ticket.getTicketId();
                case "STATE"     -> ticket.getState();
                case "CATEGORY"  -> ticket.getCategory();
                case "PRIORITY"  -> ticket.getPriority();
                case "USER"      -> ticket.getSubmittedBy().getUserId();
                case "AGENT"     -> ticket.getAssignedAgent() == null ? "UNASSIGNED" : ticket.getAssignedAgent().getUserId() ; 
                default          -> null;
            }; // compare the value to the ticket based on the given field
            if (actual != null && actual.equalsIgnoreCase(value.trim())) {
                result.add(ticket); // if yes add them
            }
        }
        return result;
    } // Returns filtered Ticket list based on the field and valuee

    public ArrayList<User> filterUsers(String field, String value) {
        ArrayList<User> result = new ArrayList<>();

        String searchItem = value.toUpperCase();
        ArrayList<User> allUsers = getAllUsers();  
        for (User user : allUsers ) { // EX ROLE = CATEGORY, VALUE = REQUESTOR
            if (user == null) { continue; }
            String actual = switch(field.toUpperCase()) {
                case "ID"             -> user.getUserId();
                case "ROLE"           -> user.getRole();
                case "NAME"           -> user.getFullName();
                case "SPECIALIZATION" -> user instanceof SupportAgent supportAgent ? supportAgent.getSpecialization() : null;
                default               -> null;
            }; // compare the value to the ticket based on the given field
            if (actual  != null && actual.toUpperCase().contains(searchItem)) {
                result.add(user); // if yes add them
            }
        }
        return result;
    } // Returns filtered User list based on the field and value

    public LinkedHashMap<String, Integer> getSummaryCounts() {
        LinkedHashMap<String, Integer> counts = new LinkedHashMap<>(); // A linked Hash map is a series of KEY-VALUE pairs EX. TOTAL-TICKETS = KEY.  15 VALUE

        counts.put("TOTAL_USERS", getAllUsers().size()); //  / USERS
       
        String[] states = {"OPEN", "IN_PROGRESS", "RESOLVED", "CLOSED", "DELETED"};
        for (String state : states) { counts.put(state, filterTickets("STATE", state).size()); } // States
        String[] priorities = {"LOW", "MEDIUM", "HIGH", "URGENT"};
        for (String priority : priorities ) { counts.put(priority, filterTickets("PRIORITY", priority).size()); } // PRIORITY
        String[] categories = {"FACILITIES_MAINTENANCE", "SOFTWARE_SYSTEMS", "NETWORK", "HARDWARE", "ACCOUNT_SECURITY"};

        return counts;
    } // this method uses the Filter Users and Filter Ticket method inputting a FIELD andf VALUE and getting the SIZE() of the lists

import java.time.LocalDateTime;

public class Comment {

    private final String commentId;
    private final User user;
    private final String message;
    private final LocalDateTime timestamp;
    private static int commentCounter = 0;

    public Comment(User user, String message) {
        this.commentId = generateCommentId();
        this.user = user;
        this.message = message;
        this.timestamp = LocalDateTime.now();
    }

    public String getCommentId() {return commentId; } 
    public User getAuthor() {return user;}
    public String getAuthorName() { return user.getFullName(); }
    public String getMessage() {return message;}
    public LocalDateTime getTimestamp() { return timestamp;}
    public String getTime() { return getTimestamp().format(java.time.format.DateTimeFormatter.ofPattern("MM-dd HH:mm")); }

    private String generateCommentId() {
        commentCounter++;
        return String.format("CM-%03d", commentCounter);
    } // called whenever a new comment is created

}

package entitylayer;

import datalayer.Repository;
import java.util.ArrayList;

public class Requestor extends User {

    public Requestor(String userId, String firstName, String lastName, String email, String password, Repository repository) {
        super("REQUESTOR",userId, firstName, lastName, email, password, repository);
    }

    @Override
    public Ticket getTicket(String ticketId) {
        Ticket ticket = getRepository().findTicketById(ticketId);
        if (ticket == null || ticket.getState().equalsIgnoreCase("DELETED")) { 
            throw new IllegalArgumentException("Ticket " + ticketId +" was not found."); 
        } // Throw an error if you try to acces a ticket that doesnt exist
        if (ticket.getSubmittedBy() == null || !ticket.getSubmittedBy().getUserId().equalsIgnoreCase(getUserId())) { 
            throw new SecurityException("You can only view your own tickets."); 
        } // Throw an Error if you try to access a ticket that isnt your
        return ticket;  
    } // throws an exception if invalid Ticket ID / if you do not own ticket

    @Override
    public ArrayList<Ticket> getTickets() {
       ArrayList<Ticket> myTickets = new ArrayList<>();
        ArrayList<Ticket> allTickets = getRepository().getAllTickets();
        if (allTickets == null ) { return myTickets; }
        for (Ticket ticket : allTickets) {
                if (ticket.getSubmittedBy().getUserId().equalsIgnoreCase(getUserId()) && !ticket.isDeleted()) {
                    myTickets.add(ticket);
                }
            }
        return myTickets;
    } // Returns all tickets submitted by the requestor except Deleted ones

    public Ticket createTicket(String priority, String category, String subject, String description) {
        return new Ticket(this, null, priority, category, subject, description);
    } // Create a ticket 

    public Comment createComment(String message){
        return new Comment(this, message);
    } // Create A comment
    
    public void submitTicket(Ticket ticket) {
        autoAssignAgent(ticket); // tickets stays unassigned if no agent matches
        getRepository().addTicket(ticket);
     } // after creating a ticket auto assign to agent and store in the repo
     
    public void deleteTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.requireActive(); // The ticket should be OPEN / IN_PROGRESS
        ticket.deleteTicket(this, reason); 
     } // Deletes a ticket if it is OPEN - IN-PROGRESS
     
    public void reopenTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.reopenTicket(this, reason);
    } // Reopens a ticket only if its RESOLVED
     
    public void addComment(String ticketId, String message) {
        Ticket ticket = getTicket(ticketId);
        ticket.addComment(createComment(message));
    }

    public void closeTicket(String ticketId, String reason) {
        Ticket ticket = getTicket(ticketId); // throws an error if not found
        ticket.closeTicket(this, reason);
    } // Close aticket that is resolved
    
    public ArrayList<Ticket> getActiveTickets() {
        ArrayList<Ticket> myTickets = new ArrayList<>();
        for (Ticket ticket : getTickets()) {
            if (ticket.isActive()) {
                myTickets.add(ticket);
            }
        }
        return myTickets;
     } // Returns all tickets submitted by the requestor that are OPEN / INPROGRESS except Deleted ones

    private void autoAssignAgent(Ticket ticket) {

        ArrayList<User> users = getRepository().getAllUsers();

        SupportAgent bestAgent = null;
        int fewestActiveTickets = Integer.MAX_VALUE;

        for (User user : users) {
            if (!(user instanceof SupportAgent agent)) { continue; } // User must be a supportAgent
            if (agent.getSpecialization() == null ) { continue; } // SupportAgent must have a specialization
            if (!agent.getSpecialization().equalsIgnoreCase(ticket.getCategory())) { continue; } // SupportAgent specializaiton must be thhe same as ticket category
            
            int activeCount = agent.getActiveTickets().size();
            if (activeCount < fewestActiveTickets) {
                fewestActiveTickets = activeCount;
                bestAgent = agent;
            } // Pick the agent with fewest active tickets that specialization matchhes the category of the ticket 
        }

        if (bestAgent != null) { 
            ticket.setAssignedAgent(bestAgent);
        } // if there is a best agent assign the ticket

        // do nothing if there isnt an agent available

    } // Picks the best agent for the job

}

    public String getSpecialization() { return specialization; }

    public void setDeskExtension(String deskExtension) { this.deskExtension = deskExtension; }

    @Override
    public Ticket getTicket(String ticketId) {
        Ticket ticket = getRepository().findTicketById(ticketId);
        if (ticket == null || ticket.getState().equalsIgnoreCase("DELETED")) {

        if (ticket.getAssignedAgent() == null || !ticket.getAssignedAgent().getUserId().equalsIgnoreCase(getUserId()) ) { 
           throw new SecurityException("You can only view tickets assigned to you.");
        } // Throw an error if the tickett isnt assigend to you

    } // Gets an existing Ticket that is assigned to you

        ArrayList<Ticket> assignedTickets = new ArrayList<>();
        ArrayList<Ticket> allTickets = getRepository().getAllTickets();
        if (allTickets == null) { return assignedTickets; }

        for (Ticket ticket : allTickets) {
            if (ticket.getAssignedAgent() != null && ticket.getAssignedAgent().getUserId().equalsIgnoreCase(getUserId()) && !ticket.isDeleted()) {
                assignedTickets.add(ticket);

        }

    } // Gets all assigned tickets that arent deleted

    public Comment createComment(String message){

    } // add a comment to a ticket

        requireActive(); // A ticket should still be OPEN / IN-PROGRESS
        
        if (user == null) {
            throw new IllegalArgumentException("This User doesnt exist");
        } // throw an error if thhe user doesnt exist

        if (!(user instanceof SupportAgent)) { 
            throw new IllegalArgumentException("User with " + user.getFullName() + " is not an agent.");
        } // Throw an error if the user isnt a supportAgent

        SupportAgent newAgent = (SupportAgent) user; // downcast User if they are asupportAgent
        
        if (assignedAgent != null && assignedAgent.equals(newAgent)) { 
            throw new IllegalArgumentException("Ticket " + ticketId + " is already assigned to " + newAgent.getFirstName() + "."); 
        } // Cannot assign to same agent

        if (!newAgent.getSpecialization().equalsIgnoreCase(category)) {
            throw new IllegalArgumentException("Ticket " + ticketId + " category ("+ getCategory() +") does not match " + newAgent.getFirstName() +" ("+ newAgent.getSpecialization() + ") specialization."); 
        } // cannot assign to an agent with a different category from the ticket

        String oldAgentName = (assignedAgent != null) ? getAssignedAgent().getFullName() : "Unassigned"; // if the there is a currently assigned agent get their name else put Unassigned

        setAssignedAgent(newAgent); // Set assigned Agent
        log(oldAgentName, newAgent.getFullName(), changedBy, reason); // Create a ticket log if successful
    } // Used by Admin. Reassign a active ticket to a different agent with the same specialization

    public boolean isResolved() { return (state.equalsIgnoreCase("RESOLVED")); } // returns if a ticket is RESOLVED
    public boolean isFinished() { return state.equals("CLOSED") || state.equals("DELETED");  } // Returns if a ticket is CLOSED / DELTED. Finished tickets should no longer be edited
    public boolean isDeleted() { return state.equals("DELETED"); } // Returns if a ticket is Deleted

    public void checkTransition(String newState) {
        boolean valid = switch(this.state) {
            case "OPEN"        -> newState.equalsIgnoreCase("IN_PROGRESS") || newState.equalsIgnoreCase("DELETED");
            case "IN_PROGRESS" -> newState.equalsIgnoreCase("OPEN") || newState.equalsIgnoreCase("RESOLVED") || newState.equalsIgnoreCase("DELETED");
            case "RESOLVED"    -> newState.equalsIgnoreCase("OPEN") || newState.equalsIgnoreCase("IN_PROGRESS") || newState.equalsIgnoreCase("CLOSED") || newState.equalsIgnoreCase("DELETED");
            case "CLOSED"      -> newState.equalsIgnoreCase("DELETED");
            default            -> false;
        }; // Compares the current state and its allowed state change EX. An open ticket is only allowed to change to IN_PROGRESS
        if (!valid) {
            throw new IllegalStateException("Ticket " + ticketId + " cannot move from " + state + " to " + newState + ".");
        }  // Throws an error if it is an invalid state transition
    } 

    public void requireNotFinished() {
        if (isFinished()) {
            throw new IllegalStateException("Ticket " + ticketId + " is " + state + " and can no longer be changed.");
        }
    } // throw and error if a ticket is already CLOSED / DELETED

    public void requireActive() {
        if (!isActive()) {
            throw new IllegalStateException("Ticket " + ticketId + " is " + state + ". Only OPEN or IN_PROGRESS tickets allow this.");
        }
    } // Throw an error if a Ticket isnt OPEN / IN-PROGRESS

    public void requireResolved() {
        if (!isResolved()) {
            throw new IllegalStateException("Ticket " + ticketId + " is " + state + ". Only RESOLVED tickets allow this.");
        }
    } // Throw an error if a ticket isnt Resolved

    public void requireNotDeleted() {
    if (isDeleted())
        throw new IllegalStateException("Ticket " + ticketId + " is already DELETED.");
    } // Throw an error if a Ticketr is Deleted

    protected int readInt(String prompt, int min, int max) {
        while (true) {
            System.out.print(prompt);
            String rawInput = input.nextLine();
            if (rawInput == null || rawInput.trim().isEmpty()) {
                System.out.println("Input cannot be blank. Please try again");
                continue;
            } 
            try {
                int value = Integer.parseInt(rawInput.trim());
                if (value >= min && value <= max) { return value; }
                System.out.println("Invalid choice. Please enter a number between " + min + " and " + max + ".");
            } catch (NumberFormatException e) {
                System.out.println("Invalid input. Please enter a valid number.");
            }
        }
    }

    protected String readLine(String prompt, boolean allowBlank) {
        while (true) {
            System.out.print(prompt);
            String line = input.nextLine();

            if (line != null) {
                line = line.trim();
                if (allowBlank || !line.isEmpty()) {
                    return line;
                }
            }
            System.out.println("Field cannot be blank. Please enter a valid value.");
        }
    } 

    protected String truncate(String text, int maxLength) {
        if (text == null) { return ""; }
        if (text.length() <= maxLength) {return text;}
        if (maxLength <= 3) { return text.substring(0, maxLength); }
        return text.substring(0, maxLength -3) + "...";
    }

            if (rawChoice == null || rawChoice.trim().isEmpty()) {
                System.out.println("Input cannot be blank. Please try again.");
                continue;
            }
            String choice = rawChoice.trim().toLowerCase();
            if (choice.equals("y") || choice.equals("yes")) {
                return true;
            } else if (choice.equals("n") || choice.equals("no")) {
                return false;
            } 
            System.out.println("Invalid Input. Please enter Y for Yes, and N for No.");
        }
    } // asks a question then returns the choice

            ticket.requireNotFinished();
            String message = readLine("Enter comment: ", false);
            supportAgent.addComment(ticketId, message);
            System.out.println("Success! Comment added to " + ticket.getTicketId() + ".");
        } catch (IllegalArgumentException | SecurityException | IllegalStateException e) {
           printError(e.getMessage());
        }
        waitForEnter();
    }

    private String readAgentState() { 
        while(true) {
            System.out.println("");
            System.out.println("[1] OPEN");
            System.out.println("[2] IN-PROGRESS");
            System.out.println("[3] RESOLVED");
            System.out.println("");
            int choice = readInt("choice: ", 1, 3);
            switch (choice) {
                case 1 -> { return "OPEN"; }
                case 2 -> { return "IN_PROGRESS"; }
                case 3 -> { return "RESOLVED"; }
            }
        }
    }
